const http = require('http');

const TARGET_URL = 'http://localhost:3000/validate';
// Create a token that expires in 10 seconds
const EXPIRY_TIME = Date.now() + 10000;
const FAKE_TOKEN = `synthetic_token_exp_${EXPIRY_TIME}`;

function sendRequest() {
  return new Promise((resolve) => {
    const req = http.request(TARGET_URL, { method: 'POST' }, (res) => {
      let data = '';
      res.on('data', chunk => data += chunk);
      res.on('end', () => resolve(data));
    });
    req.write(JSON.stringify({ token: FAKE_TOKEN }));
    req.end();
  });
}

async function runLoadTest() {
  console.log("Starting synthetic load test...");
  const promises = [];
  // Fire 500 concurrent requests just before expiry
  for (let i = 0; i < 500; i++) {
    promises.push(sendRequest());
  }
  const results = await Promise.all(promises);
  const acceptedExpired = results.filter(r => r.includes('valid')).length;

  if (acceptedExpired > 0) {
    console.log(`BUG CONFIRMED: ${acceptedExpired} expired tokens were accepted.`);
  } else {
    console.log("No race condition detected in this run.");
  }
}

runLoadTest();
