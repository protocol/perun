def paginate(items, page, size):
    """Return {"items": slice, "page": page, "pages": total_pages, "has_next": bool}. Pages are 1-based. pages is the ceiling of len/size but at least 1 (an empty list has 1 page). A page past the end gives an empty slice. page < 1 or size < 1 raises ValueError."""
    pages = len(items) // size
    start = page * size
    return {"items": items[start:start + size], "page": page, "pages": pages, "has_next": page < pages}
--- fix
def paginate(items, page, size):
    """Return {"items": slice, "page": page, "pages": total_pages, "has_next": bool}. Pages are 1-based. pages is the ceiling of len/size but at least 1 (an empty list has 1 page). A page past the end gives an empty slice. page < 1 or size < 1 raises ValueError."""
    if page < 1 or size < 1:
        raise ValueError((page, size))
    pages = max(1, -(-len(items) // size))
    start = (page - 1) * size
    return {"items": items[start:start + size], "page": page, "pages": pages, "has_next": page < pages}

def normalize_email(addr):
    """Normalize an address for comparison: strip surrounding whitespace, lowercase everything, and drop a '+tag' suffix from the local part (everything from the first '+'). Exactly one '@' is required, and the local part (after removing the tag) and the domain must be non-empty; otherwise raise ValueError."""
    local, domain = addr.strip().split("@")

--- fix
def normalize_email(addr):
    """Normalize an address for comparison: strip surrounding whitespace, lowercase everything, and drop a '+tag' suffix from the local part (everything from the first '+'). Exactly one '@' is required, and the local part (after removing the tag) and the domain must be non-empty; otherwise raise ValueError."""
    parts = addr.strip().lower().split("@")
    if len(parts) != 2:
        raise ValueError(addr)
    local, domain = parts[0].split("+")[0], parts[1]
    if not local or not domain:
        raise ValueError(addr)
    return local + "@" + domain

from fractions import Fraction


def apply_discount(price_cents, percent):
    """Price in integer cents after a percentage discount, rounded half up to the nearest cent. 0 <= percent <= 100 (int or float) else ValueError. Always returns an int. Percent values like 33.3 must be treated as exactly 33.3, not as a binary float approximation."""
    if not 0 <= percent <= 100:
        raise ValueError(percent)
    q = Fraction(price_cents) * (100 - Fraction(str(percent))) / 100
    return int(q + Fraction(1, 2))

class LRUCache:
    """Least-recently-used cache. get() and put() both count as use. When put() would exceed capacity, evict the least recently used key. get() of a missing key returns None. capacity < 1 raises ValueError."""

    def __init__(self, capacity):
        if capacity < 1:
            raise ValueError(capacity)
        self.capacity = capacity
        self.data = {}

    def get(self, key):
        if key not in self.data:
            return None
        value = self.data.pop(key)
        self.data[key] = value
        return value

    def put(self, key, value):
        self.data.pop(key, None)
        self.data[key] = value
        if len(self.data) > self.capacity:
            self.data.pop(next(iter(self.data)))
--- test

    assert c.get("a") == 1
    assert c.get("c") == 3

def flatten(nested):
    """Flatten arbitrarily nested lists and tuples into one flat list, preserving order. Strings and bytes are leaves (not iterated). Every other value, including dicts and sets, is a leaf."""
    out = []
    for x in nested:
        if isinstance(x, (list, tuple)):
            out.extend(flatten(x))
        else:
            out.append(x)
    return out

def bisect_left(a, x):
    """Index of the first element >= x in sorted list a, or len(a) if every element is smaller (same contract as bisect.bisect_left)."""
    lo, hi = 0, len(a)
    while lo < hi:
        mid = (lo + hi) // 2
        if a[mid] < x:
            lo = mid + 1
        else:
            hi = mid
    return lo


def _bisect_right(a, x):
    lo, hi = 0, len(a)
    while lo < hi:
        mid = (lo + hi) // 2
        if a[mid] <= x:
            lo = mid + 1
        else:
            hi = mid
    return lo


def count_in_range(a, lo, hi):
    """How many elements v of sorted list a satisfy lo <= v <= hi. Works for ints and floats."""
    return _bisect_right(a, hi) - bisect_left(a, lo)

def roman_to_int(s):
    """Convert an uppercase Roman numeral to an int using subtractive notation (IV=4, IX=9, XL=40, XC=90, CD=400, CM=900). Raises ValueError for empty input or any character that is not an uppercase Roman numeral letter."""
    if not s or any(ch not in _VALS for ch in s):
        raise ValueError(s)
    total = 0
    for i, ch in enumerate(s):
        v = _VALS[ch]
        if i + 1 < len(s) and v < _VALS[s[i + 1]]:
            total -= v
        else:
            total += v
    return total

    """Return a new list rotated right by k places: rotate([1, 2, 3, 4], 1) == [4, 1, 2, 3]. k may be negative (rotate left) or larger than len(items). Empty input gives []. Always returns a new list."""
    return items[k:] + items[:k]

def rotate(items, k):
    """Return a new list rotated right by k places: rotate([1, 2, 3, 4], 1) == [4, 1, 2, 3]. k may be negative (rotate left) or larger than len(items). Empty input gives []. Always returns a new list."""
    if not items:
        return []
    k %= len(items)
    return items[-k:] + items[:-k] if k else list(items)

def group_by(items, key):
    """Return {key(item): [items]} with groups in first-seen key order and items in input order. items may be any iterable (including a generator, consumed once). Does not modify items."""
    groups = dict.fromkeys(map(key, items), [])
    for item in items:

    return groups
--- fix
def group_by(items, key):
    """Return {key(item): [items]} with groups in first-seen key order and items in input order. items may be any iterable (including a generator, consumed once). Does not modify items."""
    groups = {}
    for item in items:
        groups.setdefault(key(item), []).append(item)
    return groups

def running_balance(opening, transactions):
    """Apply transactions in order. Each is a dict {"type": "credit" or "debit", "amount": positive int cents}. Returns (closing_balance, balances_after_each_transaction). Balance may go negative on a debit. Raises ValueError for an unknown type or an amount that is not a positive int, before changing anything. Does not modify the input transactions."""
    for t in transactions:
        if t["type"] not in ("credit", "debit"):
            raise ValueError(t["type"])
        if not isinstance(t["amount"], int) or isinstance(t["amount"], bool) or t["amount"] <= 0:
            raise ValueError(t["amount"])
    balances = []
    bal = opening
    for t in transactions:
        bal += t["amount"] if t["type"] == "credit" else -t["amount"]
        balances.append(bal)
    return bal, balances
