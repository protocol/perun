--- fix
def parse_csv(text, delimiter=","):
    rows, row, field = [], [], ""
    in_q = False
    fstart = True
    had = False
    i, n = 0, len(text)
    while i < n:
        c = text[i]
        if in_q:
            if c == '"':
                if text[i + 1:i + 2] == '"':
                    field += '"'
                    i += 1
                else:
                    in_q = False
            else:
                field += c
        elif c == '"' and fstart:
            in_q = True
            fstart = False
            had = True
        elif c == delimiter:
            row.append(field)
            field = ""
            fstart = True
            had = True
        elif c == "\n" or (c == "\r" and text[i + 1:i + 2] == "\n"):
            if c == "\r":
                i += 1
            if had or field or row:
                row.append(field)
                rows.append(row)
            row, field, fstart, had = [], "", True, False
        else:
            field += c
            fstart = False
            had = True
        i += 1
    if in_q:
        raise ValueError("unterminated quote")
    if had or field or row:
        row.append(field)
        rows.append(row)
    return rows

--- fix
import re


def glob_match(pattern, path):
    i, n, out = 0, len(pattern), ""
    while i < n:
        c = pattern[i]
        if c == "\\" and i + 1 < n:
            out += re.escape(pattern[i + 1])
            i += 2
        elif c == "*":
            if pattern[i:i + 3] == "**/":
                out += "(?:.*/)?"
                i += 3
            elif pattern[i:i + 2] == "**":
                out += ".*"
                i += 2
            else:
                out += "[^/]*"
                i += 1
        elif c == "?":
            out += "[^/]"
            i += 1
        elif c == "[":
            j = i + 1
            neg = pattern[j:j + 1] == "!"
            if neg:
                j += 1
            close = pattern.find("]", j)
            if close <= j:
                out += re.escape(c)
                i += 1
            else:
                body = pattern[j:close].replace("\\", "\\\\").replace("^", "\\^")
                out += ("[^/" if neg else "(?!/)[") + body + "]"
                i = close + 1
        else:
            out += re.escape(c)
            i += 1
    return re.fullmatch(out, path, re.S) is not None

set_path(obj, path, value) returns a new structure in which the location named by path holds value; obj itself and everything reachable from it must be left untouched (the result shares no mutable parts with obj). A path is dot-separated keys with optional list indexes, for example "a.b[2].c" or "items[0]". def set_path(obj, path, value):
    node = obj
    parts = path.replace("[", ".[").split(".")
    for p in parts[:-1]:
        if p.startswith("["):
            node = node[int(p[1:-1])]
        else:
            node = node[p]
    last = parts[-1]
    if last.startswith("["):
        node[int(last[1:-1])] = value
    else:
        node[last] = value
    return obj
--- fix
import copy
import re

_PATH = re.compile(r"[^.\[\]]+(?:\.[^.\[\]]+|\[\d+\])*")
_STEP = re.compile(r"\[(\d+)\]|([^.\[\]]+)")


def set_path(obj, path, value):
    if not _PATH.fullmatch(path):
        raise ValueError(path)
    steps = [int(i) if i else k for i, k in _STEP.findall(path)]
    if not isinstance(obj, dict):
        raise TypeError("obj must be a dict")
    root = copy.deepcopy(obj)
    node = root
    for n, step in enumerate(steps):
        last = n == len(steps) - 1
        fill = None if last else ([] if isinstance(steps[n + 1], int) else {})
        if isinstance(step, int):
            if not isinstance(node, list):
                raise TypeError("index step on non-list")
            while len(node) <= step:
                node.append(None)
            if last:
                node[step] = copy.deepcopy(value)
            else:
                if node[step] is None:
                    node[step] = fill
                node = node[step]
        else:
            if not isinstance(node, dict):
                raise TypeError("key step on non-dict")
            if last:
                node[step] = copy.deepcopy(value)
            else:
                if step not in node or node[step] is None:
                    node[step] = fill
                node = node[step]
    return root

        self.lots = {}

    def add(self, sku, qty, expires):

--- fix
class Inventory:
    def __init__(self):
        self.lots = {}

    def add(self, sku, qty, expires):
        if qty <= 0:
            raise ValueError("qty")
        self.lots.setdefault(sku, []).append([expires, qty])

    def _usable(self, sku, today):
        if sku not in self.lots:
            return []
        self.lots[sku] = [lot for lot in self.lots[sku] if lot[0] >= today and lot[1] > 0]
        return sorted(self.lots[sku], key=lambda lot: lot[0])

    def take(self, sku, qty, today):
        if qty <= 0:
            raise ValueError("qty")
        lots = self._usable(sku, today)
        if sum(lot[1] for lot in lots) < qty:
            raise ValueError("insufficient")
        taken = []
        for lot in lots:
            if qty == 0:
                break
            n = min(qty, lot[1])
            lot[1] -= n
            qty -= n
            taken.append((lot[0], n))
        self.lots[sku] = [lot for lot in self.lots[sku] if lot[1] > 0]
        return taken

    def stock(self, sku, today):
        return sum(lot[1] for lot in self._usable(sku, today))

--- fix
import re


def to_snake(name):
    lead, core, trail = re.match(r"^(_*)(.*?)(_*)$", name, re.S).groups()
    core = re.sub(r"([A-Z]+)([A-Z][a-z])", r"\1_\2", core)
    core = re.sub(r"([a-z0-9])([A-Z])", r"\1_\2", core)
    core = re.sub(r"[\s_-]+", "_", core)
    return lead + core.lower() + trail

### h_backoff
--- ticket
backoff_schedule(base, factor, cap, attempts, budget=None) returns the list of delays before each retry. Delay number k (starting at 0) is min(cap, base * factor**k), and there are at most `attempts` delays (0 gives []). With a budget, the running total of delays may not exceed it: once adding the next delay would pass the budget, that delay is cut down to exactly what remains and nothing follows it; if nothing remains the list stops before it. The cap applies before the budget. ValueError for attempts < 0, base <= 0, factor < 1, cap < base, or a negative budget.
Observed: the first delay is already multiplied by the factor, one attempt too few is returned, and a budget stops early instead of using the remainder.
--- src
def backoff_schedule(base, factor, cap, attempts, budget=None):
    out, total = [], 0
    for k in range(1, attempts):
        d = base * factor ** k
        if budget is not None and total + d > budget:
            break
        out.append(min(d, cap))
        total += d
    return out
--- fix
def backoff_schedule(base, factor, cap, attempts, budget=None):
    if attempts < 0 or base <= 0 or factor < 1 or cap < base or (budget is not None and budget < 0):
        raise ValueError("bad arguments")
    out, total = [], 0
    for k in range(attempts):
        d = min(cap, base * factor ** k)
        if budget is not None:
            left = budget - total
            if left <= 0:
                break
            d = min(d, left)
        out.append(d)
        total += d
    return out

import pytest
from mod import backoff_schedule


def test_plain_and_cap():
    assert backoff_schedule(1, 2, 10, 6) == [1, 2, 4, 8, 10, 10]
    assert backoff_schedule(0.5, 3, 100, 3) == [0.5, 1.5, 4.5]
    assert backoff_schedule(2, 1, 5, 3) == [2, 2, 2]
    assert backoff_schedule(1, 2, 10, 0) == []
    assert backoff_schedule(1, 2, 10, 1) == [1]


def test_budget_truncates_last_delay():
    assert backoff_schedule(1, 2, 10, 6, budget=10) == [1, 2, 4, 3]
    assert backoff_schedule(1, 2, 10, 6, budget=7) == [1, 2, 4]
    assert backoff_schedule(1, 2, 10, 6, budget=0) == []
    assert backoff_schedule(1, 2, 10, 2, budget=100) == [1, 2]
    assert backoff_schedule(4, 2, 5, 5, budget=12) == [4, 5, 3]
    assert backoff_schedule(4, 2, 5, 2, budget=1) == [1]


def test_validation():
    for kw in [dict(base=1, factor=2, cap=5, attempts=-1), dict(base=0, factor=2, cap=5, attempts=1),
               dict(base=1, factor=0.5, cap=5, attempts=1), dict(base=3, factor=2, cap=2, attempts=1),
               dict(base=1, factor=2, cap=5, attempts=1, budget=-1)]:
        with pytest.raises(ValueError):
            backoff_schedule(**kw)
