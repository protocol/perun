    """Compare dotted numeric versions: -1 if a < b, 0 if equal, 1 if a > b. Compare each part as a number ('1.10' > '1.9', '01.2' == '1.2'). Missing trailing parts count as 0 ('1.0' == '1.0.0'). Empty or non-numeric parts raise ValueError."""

def compare_versions(a, b):
    """Compare dotted numeric versions: -1 if a < b, 0 if equal, 1 if a > b. Compare each part as a number ('1.10' > '1.9', '01.2' == '1.2'). Missing trailing parts count as 0 ('1.0' == '1.0.0'). Empty or non-numeric parts raise ValueError."""
    pa, pb = [int(p) for p in a.split(".")], [int(p) for p in b.split(".")]
    n = max(len(pa), len(pb))
    pa += [0] * (n - len(pa))
    pb += [0] * (n - len(pb))
    return (pa > pb) - (pa < pb)

def truncate(text, limit):
    """Shorten text to at most `limit` characters, ellipsis included. Text that already fits is returned unchanged. Otherwise return the longest run of whole words that, followed by '...', fits in limit. If not even the first word fits, hard-cut to limit-3 characters and add '...'. limit < 4 raises ValueError."""
    if limit < 4:
        raise ValueError(limit)
    if len(text) <= limit:
        return text
    room = limit - 3
    cut = text[:room + 1].rpartition(" ")[0]
    return (cut or text[:room]) + "..."

    """Convert snake_case to lowerCamelCase. Leading and trailing underscores are preserved ('_private_name' -> '_privateName', 'name_' -> 'name_', '__init__' -> '__init__'). --- fix
def snake_to_camel(name):
    """Convert snake_case to lowerCamelCase. Leading and trailing underscores are preserved ('_private_name' -> '_privateName', 'name_' -> 'name_', '__init__' -> '__init__'). Runs of inner underscores are one separator ('a__b' -> 'aB'). Each word after the first only has its first character uppercased; the rest is untouched ('get_HTTP_code' -> 'getHTTPCode', 'x_2d' -> 'x2d'). A name made only of underscores is returned unchanged."""
    core = name.strip("_")
    if not core:
        return name
    lead = name[:len(name) - len(name.lstrip("_"))]
    trail = name[len(name.rstrip("_")):]
    words = [w for w in core.split("_") if w]
    return lead + words[0] + "".join(w[:1].upper() + w[1:] for w in words[1:]) + trail

def top_k(scores, k):
    """scores: dict name -> number. Return the k best names, highest score first; equal scores are ordered alphabetically by name. k larger than len(scores) returns all names; k <= 0 returns []."""
    ranked = sorted(scores, key=lambda n: (-scores[n], n))
    return ranked[:max(k, 0)]

class RingBuffer:
    """Fixed-capacity FIFO queue. push() appends; when full, the oldest item is overwritten. pop() removes and returns the oldest item, raising IndexError when empty. len() is the current item count. items() returns the contents oldest to newest. capacity < 1 raises ValueError."""

    def __init__(self, capacity):
        if capacity < 1:
            raise ValueError(capacity)
        self.buf = [None] * capacity
        self.head = 0
        self.size = 0

    def push(self, x):
        cap = len(self.buf)
        self.buf[(self.head + self.size) % cap] = x
        if self.size < cap:
            self.size += 1
        else:
            self.head = (self.head + 1) % cap

    def pop(self):
        if self.size == 0:
            raise IndexError("empty")
        x = self.buf[self.head]
        self.head = (self.head + 1) % len(self.buf)
        self.size -= 1
        return x

    def __len__(self):
        return self.size

    def items(self):
        return [self.buf[(self.head + i) % len(self.buf)] for i in range(self.size)]
--- test

def transpose(rows):
    """Transpose a list of equal-length rows into a list of lists. Rows of different lengths raise ValueError. transpose([]) is [] and a matrix of empty rows ([[], []]) is []."""
    if len({len(r) for r in rows}) > 1:

    return [list(col) for col in zip(*rows)]

class RateLimiter:
    """Allow at most `limit` events in any sliding window of `window` seconds. allow(now) returns True and records the event when allowed; when not allowed it returns False and records nothing. An event exactly `window` seconds old no longer counts. Timestamps never decrease."""

    def __init__(self, limit, window):
        self.limit, self.window, self.events = limit, window, []

    def allow(self, now):
        self.events = [t for t in self.events if now - t < self.window]
        if len(self.events) >= self.limit:
            return False
        self.events.append(now)
        return True

import time


def retry(fn, attempts=3, delay=0.0, exceptions=(Exception,), sleep=time.sleep):
    """Call fn() up to `attempts` times in total and return its first successful result. Only exceptions in `exceptions` trigger a retry; any other exception propagates immediately. Call sleep(delay) between attempts, but not after the last one. If every attempt fails, re-raise the last exception. attempts < 1 raises ValueError."""
    if attempts < 1:
        raise ValueError(attempts)
    for i in range(attempts):
        try:
            return fn()
        except exceptions:
            if i == attempts - 1:
                raise
            sleep(delay)

def flaky(fail_times):
    calls = []

    def fn():
        calls.append(1)
        if len(calls) <= fail_times:

        return "ok"
