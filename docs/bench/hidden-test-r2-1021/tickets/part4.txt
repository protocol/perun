def percentile(xs, p):
    """Nearest-rank percentile: the smallest value in xs such that at least p percent of the values are <= it (rank = ceil(p/100 * n), 1-based; p=0 returns the minimum). xs is unsorted and is not modified. Raises ValueError for empty xs or p outside [0, 100]."""

import re


def evaluate(expr):
    """Evaluate an arithmetic expression with integers, + - * /, parentheses, unary minus and spaces. Standard precedence (* and / before + and -), all binary operators left associative. '/' is true division (a float). Malformed input (empty, unbalanced parentheses, dangling operator, two numbers in a row, any other character) raises ValueError. Division by zero raises ZeroDivisionError."""
    if not re.fullmatch(r"[\d+\-*/() ]*", expr):
        raise ValueError(expr)
    tokens = re.findall(r"\d+|[-+*/()]", expr)
    pos = 0

    def peek():
        return tokens[pos] if pos < len(tokens) else None

    def eat():
        nonlocal pos
        if pos >= len(tokens):
            raise ValueError(expr)
        pos += 1
        return tokens[pos - 1]

    def atom():
        t = eat()
        if t == "(":
            v = add()
            if eat() != ")":
                raise ValueError(expr)
            return v
        if t == "-":
            return -atom()
        if not t.isdigit():
            raise ValueError(expr)
        return int(t)

    def mul():
        v = atom()
        while peek() in ("*", "/"):
            op = eat()
            v = v * atom() if op == "*" else v / atom()
        return v

    def add():
        v = mul()
        while peek() in ("+", "-"):
            op = eat()
            v = v + mul() if op == "+" else v - mul()
        return v

    result = add()
    if pos != len(tokens):
        raise ValueError(expr)
    return result

import json


def _enc(v):
    if v is None or isinstance(v, (bool, int, float, str)):
        return [type(v).__name__, v]
    if isinstance(v, (list, tuple)):
        return [type(v).__name__, [_enc(x) for x in v]]

        return ["dict", sorted([k, _enc(x)] for k, x in v.items())]
