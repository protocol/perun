### h_expr
--- ticket
evaluate(s) evaluates an arithmetic expression with + - * / ^, parentheses, decimal numbers and spaces. Precedence from loosest to tightest: + and - (left to right), * and / (left to right), unary minus/plus, ^ (right-associative, exponent may itself carry a sign). So -2^2 is -4, 2^3^2 is 512, -2*3 is -6, 2*-3 is -6, 2^-1 is 0.5 and --3 is 3. Division by zero raises ZeroDivisionError. Anything that is not a valid expression (empty input, unknown characters, unbalanced parentheses, a missing operand, two numbers in a row) raises ValueError.
Observed: -2^2 gives 4, 2^3^2 gives 64, and garbage like "2 $ 3" or "(2" is silently accepted.
--- src
import re


def evaluate(s):
    toks = re.findall(r"\d+\.?\d*|\.\d+|[-+*/^()]", s)
    pos = 0

    def peek():
        return toks[pos] if pos < len(toks) else None

    def take():
        nonlocal pos
        t = peek()
        pos += 1
        return t

    def expr():
        v = term()
        while peek() in ("+", "-"):
            op = take()
            r = term()
            v = v + r if op == "+" else v - r
        return v

    def term():
        v = power()
        while peek() in ("*", "/"):
            op = take()
            r = power()
            v = v * r if op == "*" else v / r
        return v

    def power():
        v = atom()
        while peek() == "^":
            take()
            v = v ** atom()
        return v

    def atom():
        t = take()
        if t == "-":
            return -atom()
        if t == "(":
            v = expr()
            take()
            return v
        return float(t)

    return expr()
--- fix
import re


def _tokens(s):
    out, i = [], 0
    while i < len(s):
        if s[i].isspace():
            i += 1
            continue
        m = re.match(r"\d+\.?\d*|\.\d+", s[i:])
        if m:
            out.append(float(m.group()))
            i += m.end()
        elif s[i] in "+-*/^()":
            out.append(s[i])
            i += 1
        else:
            raise ValueError(s[i])
    return out


def evaluate(s):
    toks = _tokens(s)
    pos = 0

    def peek():
        return toks[pos] if pos < len(toks) else None

    def take():
        nonlocal pos
        t = peek()
        pos += 1
        return t

    def expr():
        v = term()
        while peek() in ("+", "-"):
            op = take()
            r = term()
            v = v + r if op == "+" else v - r
        return v

    def term():
        v = unary()
        while peek() in ("*", "/"):
            op = take()
            r = unary()
            v = v * r if op == "*" else v / r
        return v

    def unary():
        if peek() == "-":
            take()
            return -unary()
        if peek() == "+":
            take()
            return unary()
        return power()

    def power():
        base = atom()
        if peek() == "^":
            take()
            return base ** unary()
        return base

    def atom():
        t = take()
        if isinstance(t, float):
            return t
        if t == "(":
            v = expr()
            if take() != ")":
                raise ValueError("paren")
            return v
        raise ValueError("syntax")

    v = expr()
    if pos != len(toks):
        raise ValueError("trailing")
    return v

### h_autocomplete
--- ticket
Autocomplete stores words (any characters, including punctuation, compared case-insensitively and kept lower-cased). add(word, weight=1) adds `weight` to the word's total (adding the same word again accumulates). remove(word) deletes the word and its weight and returns True, or returns False if it is not stored. suggest(prefix, k=5) returns up to k stored words that start with prefix (case-insensitive; the prefix itself counts if it is a stored word), best first: higher total weight first, equal weights in alphabetical order. k <= 0 gives []. len(ac) is the number of stored words.
Observed: re-adding a word resets its weight, equal-weight words come out in reverse alphabetical order, suggest returns k-1 words, upper-case prefixes find nothing, and words containing "$" get mixed up.
--- src
class Autocomplete:
    def __init__(self):
        self.root = {}
        self.size = 0

    def add(self, word, weight=1):
        node = self.root
        for ch in word.lower():
            node = node.setdefault(ch, {})
        if "$" not in node:
            self.size += 1
        node["$"] = weight

    def remove(self, word):
        node = self.root
        for ch in word.lower():
            if ch not in node:
                return False
            node = node[ch]
        if "$" not in node:
            return False
        del node["$"]
        self.size -= 1
        return True

    def suggest(self, prefix, k=5):
        node = self.root
        for ch in prefix:
            if ch not in node:
                return []
            node = node[ch]
        found = []

        def walk(n, acc):
            for key, child in n.items():
                if key == "$":
                    found.append((child, acc))
                else:
                    walk(child, acc + key)

        walk(node, prefix)
        found.sort(reverse=True)
        return [w for _, w in found[:k - 1]]

    def __len__(self):
        return self.size

END = object()


class Autocomplete:
    def __init__(self):
        self.root = {}
        self.size = 0

    def add(self, word, weight=1):
        node = self.root
        for ch in word.lower():
            node = node.setdefault(ch, {})
        if END not in node:
            self.size += 1
            node[END] = 0
        node[END] += weight

    def remove(self, word):
        node = self.root
        for ch in word.lower():
            if ch not in node:
                return False
            node = node[ch]
        if END not in node:
            return False
        del node[END]
        self.size -= 1
        return True

    def suggest(self, prefix, k=5):
        if k <= 0:
            return []
        prefix = prefix.lower()
        node = self.root
        for ch in prefix:
            if ch not in node:
                return []
            node = node[ch]
        found = []

        def walk(n, acc):
            for key, child in n.items():
                if key is END:
                    found.append((-child, acc))
                else:
                    walk(child, acc + key)

        walk(node, prefix)
        found.sort()
        return [w for _, w in found[:k]]

    def __len__(self):
        return self.size

### h_money_split
--- ticket
split(total, weights) divides `total` cents (a non-negative int) between people in proportion to their integer weights and returns a list of ints (same length as weights) that always sums to exactly `total`. Each person first gets floor(total * weight / sum(weights)); the cents still left over are handed out one each to the people with the largest fractional remainders, ties going to the lower index. A weight of 0 gets 0. A negative total, a negative weight, or weights that sum to 0 raise ValueError.
Observed: shares are sometimes off by a cent against each other, the last person always absorbs the rounding, and huge totals lose cents.
--- src
def split(total, weights):
    W = sum(weights)
    shares = [round(total * w / W) for w in weights]
    shares[-1] += total - sum(shares)
    return shares
--- fix
def split(total, weights):
    if total < 0 or any(w < 0 for w in weights) or sum(weights) <= 0:
        raise ValueError("bad input")
    W = sum(weights)
    shares = [total * w // W for w in weights]
    rem = [total * w % W for w in weights]
    left = total - sum(shares)
    for i in sorted(range(len(weights)), key=lambda i: (-rem[i], i))[:left]:
        shares[i] += 1
    return shares

### h_ttl_cache
--- ticket
TTLCache(capacity, ttl) is an LRU cache whose entries also expire. put(key, value, now) stores the value until now + ttl (an entry is expired when now >= stored_at + ttl); putting an existing key replaces the value, restarts its ttl and makes it the most recently used. get(key, now) returns the value, or None if missing or expired (expired entries are removed); a hit makes the key most recently used but does NOT extend its ttl. When a put needs room for a NEW key: first drop every expired entry, and only if the cache is still full evict the least recently used live entry. Updating an existing key never evicts anything. size(now) is the number of live entries. capacity <= 0 stores nothing.
Observed: an updated key evicts a neighbour, live entries are evicted while expired ones stay, size() counts dead entries, and entries are returned at the instant they expire.
--- src
from collections import OrderedDict


class TTLCache:
    def __init__(self, capacity, ttl):
        self.cap = capacity
        self.ttl = ttl
        self.d = OrderedDict()

    def get(self, key, now):
        if key not in self.d:
            return None
        v, exp = self.d[key]
        if now > exp:
            del self.d[key]
            return None
        return v

    def put(self, key, value, now):
        if len(self.d) >= self.cap:
            self.d.popitem(last=False)
        self.d[key] = (value, now + self.ttl)

    def size(self, now):
        return len(self.d)

from collections import OrderedDict


class TTLCache:
    def __init__(self, capacity, ttl):
        self.cap = capacity
        self.ttl = ttl
        self.d = OrderedDict()

    def _purge(self, now):
        for k in [k for k, (_, e) in self.d.items() if now >= e]:
            del self.d[k]

    def get(self, key, now):
        if key not in self.d:
            return None
        v, exp = self.d[key]
        if now >= exp:
            del self.d[key]
            return None
        self.d.move_to_end(key)
        return v

    def put(self, key, value, now):
        if self.cap <= 0:
            return
        self._purge(now)
        if key in self.d:
            del self.d[key]
        elif len(self.d) >= self.cap:
            self.d.popitem(last=False)
        self.d[key] = (value, now + self.ttl)

    def size(self, now):
        self._purge(now)
        return len(self.d)

from mod import TTLCache


def test_expiry_boundary_and_size():
    c = TTLCache(3, 10)
    c.put("a", 1, 0)
    c.put("b", 1, 5)

    assert c.get("a", 10) is None
    assert c.size(10) == 1
