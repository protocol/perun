def chunk(items, n):
    """Split items into consecutive lists of length n; the last may be shorter. n must be >= 1, else ValueError."""

import re

_UNITS = {"h": 3600, "m": 60, "s": 1}


def parse_duration(text):
    """Parse strings like '1h30m', '45s', '2h' into whole seconds. Units may appear in any order and may repeat ('1m1m' is 120). Whitespace is ignored. Empty or malformed text (including trailing junk such as '5x' or '1h30') raises ValueError."""
    t = text.replace(" ", "")
    if not re.fullmatch(r"(\d+[hms])+", t):
        raise ValueError(text)
    return sum(int(n) * _UNITS[u] for n, u in re.findall(r"(\d+)([hms])", t))

def merge_intervals(intervals):
    """Merge overlapping or touching [start, end] pairs (inclusive: [1, 3] and [3, 4] touch and merge, [1, 2] and [3, 4] do not). Input order is arbitrary and the input must not be modified. Returns a sorted list of [start, end] lists."""
    out = []

            out[-1][1] = max(out[-1][1], e)
        else:

def add_tag(tag, tags=None):
    """Return a new list with tag added to tags. The tags argument is never modified and nothing is shared between calls. Tags are compared case-insensitively and stored lowercased (existing tags are lowercased in the result too). A tag already present is not added twice."""
    base = [t.lower() for t in (tags or [])]
    t = tag.lower()
    return base if t in base else base + [t]

def is_leap(year):
    """Gregorian leap year: divisible by 4, except centuries, unless divisible by 400."""
    return year % 4 == 0 and (year % 100 != 0 or year % 400 == 0)


def days_in_month(year, month):
    """Days in the month (1-12). Raises ValueError for a month outside 1-12."""
    if not 1 <= month <= 12:
        raise ValueError(month)
    if month == 2:
        return 29 if is_leap(year) else 28
    return 30 if month in (4, 6, 9, 11) else 31

def median(xs):
    """Median of a non-empty list of numbers (mean of the two middle values for even length). Does not modify xs. Raises ValueError on empty input."""
    if not xs:
        raise ValueError("empty")
    s = sorted(xs)
    mid = len(s) // 2
    if len(s) % 2:
        return s[mid]
    return (s[mid - 1] + s[mid]) / 2

import re


def word_count(text):
    """Return {word: count}. A word is a run of ASCII letters, digits and apostrophes; matching is case-insensitive and keys are lowercase. "Don't" is one word. Any other character separates words."""
    counts = {}
    for w in re.findall(r"[a-z0-9']+", text.lower()):
        counts[w] = counts.get(w, 0) + 1
    return counts
